## Context

`frontend/app.py` (Streamlit) continues to run on `:8501` throughout.
This commit adds the Overview/Dashboard tab to `frontend/react/` only.
Streamlit is the reference implementation — if behaviour is unclear, check
how `with tab3:` works in `frontend/app.py`.

The Dashboard tab is the most data-rich tab. It gives the user their full financial
picture for the selected month — balance breakdown, budget health per category,
top spends, and month-over-month trends.

> Theming: follow the CSS var pattern from T2.1. See T2.3 context for the full rule.

Key improvements over Streamlit:
- Recharts stacked bar and donut chart replace custom HTML/CSS div constructions
- Animated progress bars on budget health cards (Streamlit used static `div` widths)
- MoM table is responsive with horizontal scroll (Streamlit overflow was unconstrained)
- Balance summary cards use CSS Grid — Streamlit used `st.columns([1,1,1])` which
  collapses poorly on mobile

Bonus addition (not in Streamlit): **donut chart** for category spend split.

```typescript
import { useEffect, useState, useCallback } from "react";
import { api } from "@/api/client";
import { useMonth } from "@/context/MonthContext";
import { fmtInr } from "@/utils/formatInr";
import { fmtDate } from "@/utils/formatDate";
import { CATEGORY_ICONS } from "@/utils/categories";
import { BalanceBreakdown } from "@/components/shared/BalanceBreakdown";
import { BudgetHealthCard } from "@/components/shared/BudgetHealthCard";
import { MoMTable } from "@/components/shared/MoMTable";
import { SpendDonut } from "@/components/shared/SpendDonut";
import type { Summary, ProjectionItem } from "@/types";

interface TopSpend {
  vendor: string;
  amount: number;
  category: string;
  date: string;
  note: string | null;
}
interface MoM {
  months: string[];
  categories: Record<string, Record<string, number>>;
}

export function OverviewTab() {
  const { selMonth } = useMonth();

  const [summary, setSummary]         = useState<Summary | null>(null);
  const [projections, setProjections] = useState<ProjectionItem[]>([]);
  const [topSpends, setTopSpends]     = useState<TopSpend[]>([]);
  const [mom, setMom]                 = useState<MoM | null>(null);
  const [loading, setLoading]         = useState(true);

  const load = useCallback(async () => {
    setLoading(true);
    const [sum, proj, top, momData] = await Promise.all([
      api.get<Summary>(`/summary/${selMonth}`).then(r => r.data),
      api.get<ProjectionItem[]>(`/insights/projection/${selMonth}`).then(r => r.data),
      api.get<TopSpend[]>(`/insights/top-spends/${selMonth}?limit=5`).then(r => r.data),
      api.get<MoM>(`/insights/mom/${selMonth}`).then(r => r.data),
    ]);
    setSummary(sum);
    setProjections(proj);
    setTopSpends(top);
    setMom(momData);
    setLoading(false);
  }, [selMonth]);

  useEffect(() => { load(); }, [load]);

  if (loading) return <OverviewSkeleton />;

  if (!summary) {
    return (
      <div className="text-center py-16">
        <p className="text-white/30 text-sm">
          No data yet this month.<br />
          Head to <span className="text-indigo-400">Today</span> to log expenses or{" "}
          <span className="text-indigo-400">Settings → My Take-home</span> to record income.
        </p>
      </div>
    );
  }

  const { balance } = summary;

  return (
    <div className="space-y-6">

      {/* ── Balance Summary Cards ───────────────────────── */}
      <section>
        {/* 3-card grid — responsive on mobile */}
        <div className="grid grid-cols-3 gap-3 mb-4">
          <BalanceSummaryCard
            label="Remaining"
            value={balance.remaining}
            sub="After all paid expenses"
            accent
            highlight={balance.remaining >= 0 ? "success" : "danger"}
          />
          <BalanceSummaryCard
            label="Income"
            value={balance.total_income}
            sub={`${fmtInr(balance.variable_total)} variable`}
          />
          <BalanceSummaryCard
            label="Fixed"
            value={balance.fixed_paid_total}
            sub={`${fmtInr(balance.fixed_unpaid_total)} pending`}
          />
        </div>

        {/* Balance breakdown stacked bar — replaces Streamlit custom HTML div */}
        <BalanceBreakdown balance={balance} />
      </section>

      {/* ── Category Donut (bonus — not in Streamlit) ───── */}
      {summary.categories.length > 0 && (
        <section>
          <h2 className="text-xs font-syne font-bold uppercase tracking-widest text-white/40 mb-3">
            Spend by Category
          </h2>
          <SpendDonut categories={summary.categories} />
        </section>
      )}

      {/* ── Budget Health ────────────────────────────────── */}
      {projections.length > 0 && (
        <section>
          <h2 className="text-xs font-syne font-bold uppercase tracking-widest text-white/40 mb-3">
            Budget Health
          </h2>
          <div className="space-y-3">
            {projections.map(p => (
              <BudgetHealthCard key={p.category} projection={p} />
            ))}
          </div>
        </section>
      )}

      {/* ── Top Spends ──────────────────────────────────── */}
      {topSpends.length > 0 && (
        <section>
          <h2 className="text-xs font-syne font-bold uppercase tracking-widest text-white/40 mb-3">
            Top Spends This Month
          </h2>
          <div className="space-y-1">
            {topSpends.map((t, i) => (
              <TopSpendRow key={i} rank={i + 1} item={t} />
            ))}
          </div>
        </section>
      )}

      {/* ── Month-over-Month ─────────────────────────────── */}
      {mom && mom.months.length > 0 && (
        <section>
          <h2 className="text-xs font-syne font-bold uppercase tracking-widest text-white/40 mb-3">
            Month-over-Month
          </h2>
          <MoMTable mom={mom} />
        </section>
      )}

    </div>
  );
}
```

### `src/components/shared/BalanceBreakdown.tsx`

Replaces Streamlit's custom HTML multi-segment div.

```typescript
import { fmtInr } from "@/utils/formatInr";

interface Balance {
  total_income: number;
  fixed_paid_total: number;
  fixed_unpaid_total: number;
  variable_total: number;
  remaining: number;
}

export function BalanceBreakdown({ balance }: { balance: Balance }) {
  const inc = balance.total_income || 1; // avoid divide by zero

  const segments = [
    { label: "Fixed Paid",  value: balance.fixed_paid_total,   colour: "#6366f1"              },
    { label: "Pending",     value: balance.fixed_unpaid_total,  colour: "rgba(99,102,241,0.3)" },
    { label: "Variable",    value: balance.variable_total,      colour: "#f87171"              },
    { label: "Remaining",   value: Math.max(balance.remaining, 0), colour: "rgba(52,211,153,0.4)" },
  ];

  return (
    <div className="bg-dark-card border border-white/10 rounded-2xl p-4">
      <p className="text-white/40 text-xs uppercase tracking-widest mb-3">Monthly Breakdown</p>

      {/* Stacked bar */}
      <div className="flex h-7 rounded-lg overflow-hidden gap-0.5">
        {segments.map(s => {
          const pct = Math.min((s.value / inc) * 100, 100);
          if (pct < 0.5) return null;
          return (
            <div
              key={s.label}
              className="flex items-center justify-center text-white text-[10px] font-semibold transition-all duration-700"
              style={{ width: `${pct}%`, background: s.colour, minWidth: 0 }}
            >
              {pct > 8 ? `${Math.round(pct)}%` : ""}
            </div>
          );
        })}
      </div>

      {/* Legend */}
      <div className="flex flex-wrap gap-x-4 gap-y-1 mt-3">
        {segments.map(s => (
          <span key={s.label} className="flex items-center gap-1.5 text-white/50 text-xs">
            <span className="w-2.5 h-2.5 rounded-sm flex-shrink-0" style={{ background: s.colour }} />
            {s.label} {fmtInr(s.value)}
          </span>
        ))}
      </div>
    </div>
  );
}
```

### `src/components/shared/SpendDonut.tsx`

Bonus component not in Streamlit — proper donut chart using Recharts.

```typescript
import { PieChart, Pie, Cell, Tooltip, ResponsiveContainer } from "recharts";
import { CATEGORY_ICONS } from "@/utils/categories";
import { fmtInr } from "@/utils/formatInr";

const CHART_COLOURS = [
  "#6366f1","#8b5cf6","#ec4899","#f59e0b","#34d399","#06b6d4","#f87171","#a3e635",
];

interface Props { categories: Array<{ category: string; spent: number }>; }

export function SpendDonut({ categories }: Props) {
  const data = categories.filter(c => c.spent > 0).slice(0, 8);
  if (data.length === 0) return null;

  return (
    <div className="bg-dark-card border border-white/10 rounded-2xl p-4">
      <ResponsiveContainer width="100%" height={180}>
        <PieChart>
          <Pie
            data={data}
            dataKey="spent"
            nameKey="category"
            cx="50%"
            cy="50%"
            innerRadius={50}
            outerRadius={80}
            paddingAngle={2}
          >
            {data.map((_, i) => (
              <Cell key={i} fill={CHART_COLOURS[i % CHART_COLOURS.length]} />
            ))}
          </Pie>
          <Tooltip
            formatter={(v: number) => fmtInr(v)}
            contentStyle={{ background: "#1a1a28", border: "1px solid rgba(255,255,255,0.1)", borderRadius: 12 }}
            labelStyle={{ color: "white" }}
          />
        </PieChart>
      </ResponsiveContainer>

      {/* Category legend */}
      <div className="grid grid-cols-2 gap-x-4 gap-y-1.5 mt-2">
        {data.map((c, i) => (
          <div key={c.category} className="flex items-center gap-2 text-xs">
            <span className="w-2 h-2 rounded-full flex-shrink-0" style={{ background: CHART_COLOURS[i % CHART_COLOURS.length] }} />
            <span className="text-white/60 truncate">{CATEGORY_ICONS[c.category]} {c.category}</span>
            <span className="text-white/40 ml-auto">{fmtInr(c.spent)}</span>
          </div>
        ))}
      </div>
    </div>
  );
}
```

```typescript
import { fmtInr } from "@/utils/formatInr";
import { CATEGORY_ICONS } from "@/utils/categories";
import type { ProjectionItem } from "@/types";

const STATUS_CONFIG = {
  over:    { dot: "🔴", accent: "#ef4444", bg: "rgba(239,68,68,0.08)",   label: "Over budget"  },
  danger:  { dot: "🟠", accent: "#f59e0b", bg: "rgba(245,158,11,0.07)",  label: ""             },
  warning: { dot: "🟡", accent: "#eab308", bg: "rgba(234,179,8,0.06)",   label: ""             },
  safe:    { dot: "🟢", accent: "#34d399", bg: "rgba(52,211,153,0.05)",  label: "On track"     },
};

export function BudgetHealthCard({ projection: p }: { projection: ProjectionItem }) {
  const cfg = STATUS_CONFIG[p.status] ?? STATUS_CONFIG.safe;
  const barW = Math.min(p.pct_spent, 100);
  const label = cfg.label || `Projected ${fmtInr(p.projected)}`;
  const daysInfo = p.days_left > 0
    ? `${p.days_left}d left · ${fmtInr(p.daily_rate)}/day`
    : "Month complete";

  return (
    <div
      className="rounded-2xl p-4 border-l-4"
      style={{ background: cfg.bg, borderColor: cfg.accent }}
    >
      <div className="flex items-center justify-between mb-2">
        <span className="text-white font-semibold text-sm">
          {cfg.dot} {CATEGORY_ICONS[p.category] ?? "📦"} {p.category}
        </span>
        <span className="text-xs font-semibold" style={{ color: cfg.accent }}>{label}</span>
      </div>

      {/* Animated progress bar */}
      <div className="h-1.5 bg-white/5 rounded-full mb-2 overflow-hidden">
        <div
          className="h-full rounded-full transition-all duration-700"
          style={{ width: `${barW}%`, background: cfg.accent }}
        />
      </div>

      <div className="flex justify-between text-xs text-white/40">
        <span>{fmtInr(p.spent)} spent of {fmtInr(p.limit)}</span>
        <span>{daysInfo}</span>
      </div>
    </div>
  );
}
```

### `src/components/shared/MoMTable.tsx`

Responsive table with horizontal scroll — Streamlit's table could not scroll horizontally.

```typescript
import { fmtInr } from "@/utils/formatInr";
import { CATEGORY_ICONS } from "@/utils/categories";

interface MoM {
  months: string[];
  categories: Record<string, Record<string, number>>;
}

function fmtMonthLabel(m: string) {
  const [y, mo] = m.split("-");
  return new Date(Number(y), Number(mo) - 1).toLocaleDateString("en-IN", { month: "short", year: "numeric" });
}

export function MoMTable({ mom }: { mom: MoM }) {
  const { months, categories } = mom;

  return (
    <div className="bg-dark-card border border-white/10 rounded-2xl overflow-hidden">
      <div className="overflow-x-auto">
        <table className="w-full text-sm border-collapse">
          <thead>
            <tr className="border-b border-white/10">
              <th className="text-left text-white/40 text-xs px-4 py-2 font-semibold">Category</th>
              {months.map(m => (
                <th key={m} className="text-right text-white/40 text-xs px-4 py-2 font-semibold whitespace-nowrap">
                  {fmtMonthLabel(m)}
                </th>
              ))}
              <th className="text-center text-white/40 text-xs px-4 py-2 font-semibold">Trend</th>
            </tr>
          </thead>
          <tbody>
            {Object.entries(categories).sort().map(([cat, monthData]) => {
              const vals = months.map(m => monthData[m] ?? 0);
              const last = vals[vals.length - 1];
              const prev = vals[vals.length - 2];
              const chg = prev > 0 ? ((last - prev) / prev) * 100 : null;

              const trend =
                chg === null ? <span className="text-white/30">—</span>
                : chg > 10   ? <span className="text-red-400">↑ {Math.round(chg)}%</span>
                : chg < -10  ? <span className="text-emerald-400">↓ {Math.round(Math.abs(chg))}%</span>
                :               <span className="text-white/30">→</span>;

              return (
                <tr key={cat} className="border-b border-white/5">
                  <td className="px-4 py-2 text-white text-xs whitespace-nowrap">
                    {CATEGORY_ICONS[cat] ?? "📦"} {cat}
                  </td>
                  {vals.map((v, i) => {
                    const isLatest = i === vals.length - 1;
                    const isPeak   = v === Math.max(...vals) && v > 0 && isLatest;
                    return (
                      <td
                        key={i}
                        className={`px-4 py-2 text-right font-syne text-xs whitespace-nowrap ${
                          isLatest ? "text-white font-bold" : "text-white/40"
                        } ${isPeak ? "bg-red-500/10" : ""}`}
                      >
                        {v > 0 ? fmtInr(v) : "—"}
                      </td>
                    );
                  })}
                  <td className="px-4 py-2 text-center text-xs">{trend}</td>
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
}
```
