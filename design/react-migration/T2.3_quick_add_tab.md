## Context

`frontend/app.py` (Streamlit) continues to run on `:8501` throughout.
This commit adds the Quick Add tab to `frontend/react/` only.
Streamlit is the reference implementation — if behaviour is unclear, check
how `with tab1:` works in `frontend/app.py`.

> **Theming rule (applies to ALL tab components T2.3–T2.8):**
> The CSS var theming system is fully set up in `index.css` from T2.1.
> Never use `bg-dark-bg` on an outer wrapper div — use `style={{ backgroundColor: 'var(--bg)' }}`.
> `text-white`, `text-white/40`, `bg-dark-card2`, `border-white/10` etc. all respond
> to theme toggling automatically via `index.css` class remapping.
> Use `style={{ color: 'var(--text-sub)' }}` for secondary text that needs precise control.

The Streamlit version (`tab1` in `frontend/app.py`) has three sections:

Key improvements over Streamlit:
- Native keyboard focus on input when tab opens (Streamlit can't auto-focus)
- Swipe-to-delete on mobile via touch events (Streamlit used fragile JS injection)
- Animated expense cards after save (Streamlit rerenders the entire page)
- Horizontal scroll chip row for favourites (Streamlit used 3-column grid)

## Files to Create

### `src/components/tabs/QuickAddTab.tsx`

```typescript
import { useState, useRef, useEffect } from "react";
import { api } from "@/api/client";
import { useMonth } from "@/context/MonthContext";
import { CATEGORY_ICONS } from "@/utils/categories";
import { fmtInr } from "@/utils/formatInr";
import { fmtDate } from "@/utils/formatDate";
import type { Expense, ExpenseTemplate, Summary } from "@/types";
import { Loader2, Trash2, Zap } from "lucide-react";

interface ParseResult {
  saved: Expense[];
  warnings: Array<{ level: string; message: string }>;
  balance: { remaining: number };
}

export function QuickAddTab() {
  const { selMonth } = useMonth();
  const today = new Date().toISOString().slice(0, 10);

  const [text, setText] = useState("");
  const [date, setDate] = useState(today);
  const [parsing, setParsing] = useState(false);
  const [lastResult, setLastResult] = useState<ParseResult | null>(null);

  const [favourites, setFavourites] = useState<ExpenseTemplate[]>([]);
  const [todayExpenses, setTodayExpenses] = useState<Expense[]>([]);
  const [loadingFav, setLoadingFav] = useState<number | null>(null);

  const inputRef = useRef<HTMLInputElement>(null);

  // Auto-focus input when tab mounts
  useEffect(() => { inputRef.current?.focus(); }, []);

  // Load favourites + today's entries
  useEffect(() => {
    api.get<ExpenseTemplate[]>("/expense-templates").then(r => setFavourites(r.data));
    refreshToday();
  }, [selMonth]);

  const refreshToday = () => {
    api.get<Expense[]>(`/expenses/${selMonth}`).then(r => {
      setTodayExpenses(
        r.data.filter(e => e.date === today && !e.is_fixed).slice(0, 10)
      );
    });
  };

  const handleParse = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!text.trim()) return;
    setParsing(true);
    setLastResult(null);
    try {
      const { data } = await api.post<ParseResult>("/expenses/parse", {
        text,
        date_override: date,
      });
      setLastResult(data);
      setText("");
      refreshToday();
    } finally {
      setParsing(false);
    }
  };

  const handleFavLog = async (id: number) => {
    setLoadingFav(id);
    try {
      await api.post(`/expense-templates/${id}/log`);
      refreshToday();
    } finally {
      setLoadingFav(null);
    }
  };

  const handleDelete = async (id: number) => {
    await api.delete(`/expenses/${id}`);
    setTodayExpenses(prev => prev.filter(e => e.id !== id));
  };

  return (
    <div className="space-y-6">

      {/* ── NL Input Form ───────────────────────────────── */}
      <section>
        <h2 className="text-xs font-syne font-bold uppercase tracking-widest text-white/40 mb-3">
          Log Expenses
        </h2>
        <p className="text-white/40 text-xs mb-3">
          Type like:{" "}
          <code className="bg-dark-card2 px-2 py-0.5 rounded text-indigo-300">
            zomato 500, ola 200, bigbasket 1200
          </code>
        </p>

        <form onSubmit={handleParse} className="space-y-3">
          <input
            ref={inputRef}
            value={text}
            onChange={e => setText(e.target.value)}
            placeholder="e.g. zomato 500, ola 200, bigbasket 1200"
            className="w-full bg-dark-card2 border border-white/10 rounded-xl px-4 py-3 text-white placeholder-white/30 focus:border-accent focus:outline-none"
          />
          <div className="flex gap-3">
            <input
              type="date"
              value={date}
              onChange={e => setDate(e.target.value)}
              className="bg-dark-card2 border border-white/10 rounded-xl px-3 py-2 text-white text-sm"
            />
            <button
              type="submit"
              disabled={parsing || !text.trim()}
              className="flex-1 flex items-center justify-center gap-2 bg-gradient-to-r from-accent to-accent2 text-white font-syne font-semibold py-2.5 rounded-xl disabled:opacity-50"
            >
              {parsing ? <Loader2 size={16} className="animate-spin" /> : <Zap size={16} />}
              {parsing ? "Parsing…" : "Add Expenses"}
            </button>
          </div>
        </form>

        {/* Parse result cards */}
        {lastResult && (
          <div className="mt-4 space-y-3">
            {lastResult.warnings.map((w, i) => (
              <div
                key={i}
                className={`p-3 rounded-xl text-sm border-l-4 ${
                  w.level === "danger"
                    ? "bg-red-500/10 border-red-500 text-red-300"
                    : "bg-amber-500/10 border-amber-500 text-amber-300"
                }`}
              >
                {w.message}
              </div>
            ))}
            <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
              {lastResult.saved.map(item => (
                <div
                  key={item.id}
                  className="bg-dark-card border border-white/10 rounded-2xl p-4 text-center animate-fade-in"
                >
                  <div className="text-2xl mb-1">{CATEGORY_ICONS[item.category] ?? "📦"}</div>
                  <div className="text-white font-medium text-sm truncate">{item.vendor}</div>
                  <div className="font-syne font-bold text-red-400 text-lg">{fmtInr(item.amount)}</div>
                  <div className="text-white/40 text-xs">{item.category}</div>
                </div>
              ))}
            </div>
            {lastResult.balance && (
              <div className="bg-dark-card border border-white/10 rounded-2xl p-4 text-center">
                <p className="text-white/40 text-xs mb-1">Updated Remaining Balance</p>
                <p className={`font-syne font-bold text-2xl ${
                  lastResult.balance.remaining >= 0 ? "text-emerald-400" : "text-red-400"
                }`}>
                  {fmtInr(lastResult.balance.remaining)}
                </p>
              </div>
            )}
          </div>
        )}
      </section>

      {/* ── Favourites ──────────────────────────────────── */}
      {favourites.length > 0 && (
        <section>
          <h2 className="text-xs font-syne font-bold uppercase tracking-widest text-white/40 mb-2">
            Quick Add Favourites
          </h2>
          <p className="text-white/40 text-xs mb-3">Tap to log instantly</p>
          {/* Horizontal scroll — better than Streamlit's fixed 3-col grid on mobile */}
          <div className="flex gap-2 overflow-x-auto pb-2 -mx-4 px-4 scrollbar-hide">
            {favourites.map(tmpl => (
              <button
                key={tmpl.id}
                onClick={() => handleFavLog(tmpl.id)}
                disabled={loadingFav === tmpl.id}
                className="flex-shrink-0 flex flex-col items-center gap-1 bg-dark-card2 border border-white/10 rounded-2xl px-4 py-3 min-w-[80px] disabled:opacity-50"
              >
                <span className="text-xl">{CATEGORY_ICONS[tmpl.category] ?? "📦"}</span>
                <span className="text-white text-xs font-medium">{tmpl.name}</span>
                <span className="text-red-400 font-syne font-semibold text-xs">{fmtInr(tmpl.amount)}</span>
              </button>
            ))}
          </div>
        </section>
      )}

      {/* ── Today's Entries ─────────────────────────────── */}
      <section>
        <h2 className="text-xs font-syne font-bold uppercase tracking-widest text-white/40 mb-3">
          Today's Entries
        </h2>
        {todayExpenses.length === 0 ? (
          <p className="text-white/30 text-sm py-6 text-center">
            Nothing logged today.<br />
            Type something like: <span className="text-indigo-400">zomato 350, ola 120</span>
          </p>
        ) : (
          <div className="space-y-1">
            {todayExpenses.map(e => (
              <ExpenseRowSwipeable
                key={e.id}
                expense={e}
                onDelete={() => handleDelete(e.id)}
              />
            ))}
          </div>
        )}
      </section>

    </div>
  );
}
```

### `src/components/shared/ExpenseRowSwipeable.tsx`

Swipe-to-delete on mobile. In Streamlit this was injected JS that was fragile —
in React it's clean touch events.

```typescript
import { useRef, useState } from "react";
import { Trash2 } from "lucide-react";
import { CATEGORY_ICONS } from "@/utils/categories";
import { fmtInr } from "@/utils/formatInr";
import type { Expense } from "@/types";

interface Props { expense: Expense; onDelete: () => void; }

export function ExpenseRowSwipeable({ expense, onDelete }: Props) {
  const [offsetX, setOffsetX] = useState(0);
  const startX = useRef(0);
  const THRESHOLD = 80;

  const onTouchStart = (e: React.TouchEvent) => {
    startX.current = e.touches[0].clientX;
  };
  const onTouchMove = (e: React.TouchEvent) => {
    const dx = e.touches[0].clientX - startX.current;
    if (dx < 0) setOffsetX(Math.max(dx, -THRESHOLD));
  };
  const onTouchEnd = () => {
    if (offsetX <= -THRESHOLD) {
      onDelete();
    } else {
      setOffsetX(0);
    }
  };

  const icon = CATEGORY_ICONS[expense.category] ?? "📦";

  return (
    <div className="relative overflow-hidden rounded-xl">
      {/* Delete reveal layer */}
      <div className="absolute inset-y-0 right-0 w-20 bg-red-500 flex items-center justify-center rounded-r-xl">
        <Trash2 size={20} className="text-white" />
      </div>

      {/* Expense row — slides left on swipe */}
      <div
        className="relative bg-dark-card border border-white/5 rounded-xl flex items-center gap-3 px-4 py-3 transition-transform"
        style={{ transform: `translateX(${offsetX}px)` }}
        onTouchStart={onTouchStart}
        onTouchMove={onTouchMove}
        onTouchEnd={onTouchEnd}
      >
        <div className="w-9 h-9 rounded-xl bg-dark-card2 flex items-center justify-center text-lg flex-shrink-0">
          {icon}
        </div>
        <div className="flex-1 min-w-0">
          <p className="text-white text-sm font-medium truncate">{expense.vendor}</p>
          <p className="text-white/40 text-xs">{expense.category}</p>
        </div>
        <span className="font-syne font-semibold text-red-400 flex-shrink-0">
          -{fmtInr(expense.amount)}
        </span>
        {/* Desktop delete button — hidden on mobile where swipe is used */}
        <button
          onClick={onDelete}
          className="hidden sm:flex w-7 h-7 items-center justify-center rounded-lg text-white/30 hover:text-red-400 hover:bg-red-500/10 flex-shrink-0"
        >
          <Trash2 size={14} />
        </button>
      </div>
    </div>
  );
}
```
