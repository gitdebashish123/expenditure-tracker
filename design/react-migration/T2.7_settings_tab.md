## Context

`frontend/app.py` (Streamlit) continues to run on `:8501` throughout.
This commit adds the Settings tab to `frontend/react/` only.
Streamlit is the reference implementation — if behaviour is unclear, check
how `with tab5:` works in `frontend/app.py`.

Settings is the most content-heavy tab — 5 sections covering income, bills,
spending caps, shortcuts, and data export. My Account is moved to a separate
`/account` route (done in T2.8).

> Theming: follow the CSS var pattern from T2.1. See T2.3 context for the full rule.

Key improvements over Streamlit:
- Add Bill form: "No, it varies" caption shows instantly on radio change
- Budget caps: inline number inputs update with live spent-context colour
- Fixed templates grouped by category with expandable sections
- CSV export buttons with loading state (Streamlit had no loading state on download buttons)

```typescript
import { useEffect, useState } from "react";
import { api } from "@/api/client";
import { useMonth } from "@/context/MonthContext";
import type { IncomeEntry } from "@/types";
import { Save } from "lucide-react";

export function IncomeSection() {
  const { selMonth, isCurrent } = useMonth();

  const [source, setSource] = useState("Salary");
  const [amount, setAmount] = useState(0);
  const [note, setNote]     = useState("");
  const [saving, setSaving] = useState(false);

    setTimeout(() => setSaved(false), 2000);
  };

  return (
    <section>
      <div className="mb-4">
        <h2 className="font-syne font-bold text-white">💰 My Take-home</h2>
        <p className="text-white/40 text-sm mt-0.5">

        </p>
        <div className="border-b border-white/10 mt-3" />
      </div>

            value={source} onChange={e => setSource(e.target.value)}
```

### `src/components/settings/BillsSection.tsx`

This is the largest settings section — fixed templates, pool templates, and the Add Bill form.

```typescript
import { useEffect, useState, useCallback } from "react";
import { api } from "@/api/client";
import { CATEGORY_ICONS, FIXED_CATEGORIES } from "@/utils/categories";
import { fmtInr } from "@/utils/formatInr";
import type { FixedExpenseTemplate } from "@/types";
import { ChevronDown, ChevronUp, Trash2, Save, Plus } from "lucide-react";

export function BillsSection() {
  const [templates, setTemplates]   = useState<FixedExpenseTemplate[]>([]);
  const [addOpen, setAddOpen]       = useState(false);

  // Add bill form state
  const [newName, setNewName]       = useState("");
  const [newCat, setNewCat]         = useState(FIXED_CATEGORIES[0]);
  const [newKind, setNewKind]       = useState<"fixed" | "pool">("fixed");
  const [newAmt, setNewAmt]         = useState(0);
  const [addError, setAddError]     = useState<string | null>(null);
  const [adding, setAdding]         = useState(false);

  const load = useCallback(() => {
    api.get<FixedExpenseTemplate[]>("/fixed-templates").then(r => setTemplates(r.data));
  }, []);

  useEffect(() => { load(); }, [load]);

  const activeFixed = templates.filter(t => t.is_active && t.template_type === "fixed");
  const activePools = templates.filter(t => t.is_active && t.template_type === "pool");

  // Group fixed by category
  const byCat = activeFixed.reduce<Record<string, FixedExpenseTemplate[]>>((acc, t) => {
    (acc[t.category] ??= []).push(t);
    return acc;
  }, {});

  const handleAddBill = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!newName.trim()) { setAddError("Please enter a bill name."); return; }
    if (newKind === "fixed" && newAmt <= 0) { setAddError("Please enter the monthly amount for fixed bills."); return; }
    setAddError(null); setAdding(true);
    await api.post("/fixed-templates", { name: newName, category: newCat, amount: newAmt, template_type: newKind });
    setNewName(""); setNewAmt(0); setNewKind("fixed");
    setAddOpen(false); setAdding(false);
    load();
  };

  const handleDelete = async (id: number) => {
    await api.delete(`/fixed-templates/${id}`);
    setTemplates(prev => prev.filter(t => t.id !== id));
  };

  return (
    <section>
      <div className="mb-4">
        <h2 className="font-syne font-bold text-white">📋 Monthly Bills</h2>
        <p className="text-white/40 text-sm mt-0.5">
          Fixed = same every month. Variable Recurring = amount changes (like electric bill).
        </p>
        <div className="border-b border-white/10 mt-3" />
      </div>

      {/* Fixed bills grouped by category */}
      {Object.keys(byCat).length > 0 && (
        <div className="mb-4">
          <p className="text-white/40 text-xs font-semibold uppercase tracking-widest mb-3">Same amount every month</p>
          <div className="space-y-2">
            {Object.entries(byCat).sort().map(([cat, items]) => (
              <FixedTemplateCategoryGroup
                key={cat}
                category={cat}
                items={items}
                onDelete={handleDelete}
                onSaved={load}
              />
            ))}
          </div>
        </div>
      )}

      {/* Pool templates */}
      {activePools.length > 0 && (
        <div className="mb-4">
          <p className="text-white/40 text-xs font-semibold uppercase tracking-widest mb-3">Amount changes each month</p>
          <div className="space-y-2">
            {activePools.map(t => (
              <div key={t.id} className="flex items-center justify-between bg-dark-card2 border border-white/10 rounded-xl px-4 py-3">
                <div>
                  <span className="text-white text-sm font-medium">{CATEGORY_ICONS[t.category] ?? "📦"} {t.name}</span>
                  <span className="text-white/40 text-xs ml-2">{t.category} · Add payments in Fixed tab</span>
                </div>
                <button onClick={() => handleDelete(t.id)} className="text-white/20 hover:text-red-400">
                  <Trash2 size={14} />
                </button>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Add new bill expandable */}
      <div className="border border-white/10 rounded-2xl overflow-hidden">
        <button
          onClick={() => setAddOpen(o => !o)}
          className="w-full flex items-center justify-between px-4 py-3 text-white/60 hover:text-white text-sm"
        >
          <span className="flex items-center gap-2"><Plus size={14} /> Add a new bill</span>
          {addOpen ? <ChevronUp size={14} /> : <ChevronDown size={14} />}
        </button>

        {addOpen && (
          <form onSubmit={handleAddBill} className="px-4 pb-4 space-y-3 border-t border-white/10">
            <div className="grid grid-cols-2 gap-3 pt-3">
              <input
                value={newName} onChange={e => setNewName(e.target.value)}
                placeholder="e.g. Rent, Car Loan, Netflix"
                className="col-span-2 bg-dark-bg border border-white/10 rounded-xl px-3 py-2.5 text-white text-sm placeholder-white/30"
              />
              <select
                value={newCat} onChange={e => setNewCat(e.target.value)}
                className="bg-dark-bg border border-white/10 rounded-xl px-3 py-2.5 text-white text-sm"
              >
                {FIXED_CATEGORIES.map(c => <option key={c} value={c}>{CATEGORY_ICONS[c]} {c}</option>)}
              </select>
              <input
                type="number" min="0" step="100"
                value={newAmt || ""}
                onChange={e => setNewAmt(Number(e.target.value))}
                placeholder={newKind === "fixed" ? "Monthly amount (₹)" : "Typical amount (₹)"}
                className="bg-dark-bg border border-white/10 rounded-xl px-3 py-2.5 text-white text-sm"
              />
            </div>

            {/* Fixed / variable radio */}
            <div>
              <p className="text-white/50 text-xs mb-2">Is the amount the same every month?</p>
              {(["fixed", "pool"] as const).map(kind => (
                <label key={kind} className="flex items-center gap-2 mb-1.5 cursor-pointer">
                  <input type="radio" name="newKind" value={kind}
                    checked={newKind === kind} onChange={() => setNewKind(kind)}
                    className="accent-indigo-500" />
                  <span className="text-white text-sm">
                    {kind === "fixed" ? "Yes, always the same" : "No, it varies"}
                  </span>
                </label>
              ))}
              {/* Caption shows instantly on radio change — unlike Streamlit */}
              {newKind === "pool" && (
                <p className="text-white/40 text-xs mt-1">
                  You'll add the actual amount once it's paid.
                </p>
              )}
            </div>

            {addError && <p className="text-red-400 text-xs">{addError}</p>}

            <button
              type="submit" disabled={adding}
              className="w-full bg-gradient-to-r from-accent to-accent2 text-white font-semibold py-2.5 rounded-xl text-sm disabled:opacity-50"
            >
              {adding ? "Adding…" : "＋ Add Bill"}
            </button>
          </form>
        )}
      </div>
    </section>
  );
}

/* ── Fixed template category group with inline edit ─────────── */
function FixedTemplateCategoryGroup({
  category, items, onDelete, onSaved
}: {
  category: string;
  items: FixedExpenseTemplate[];
  onDelete: (id: number) => void;
  onSaved: () => void;
}) {
  const [expanded, setExpanded] = useState(false);
  const icon = CATEGORY_ICONS[category] ?? "📦";

  return (
    <div className="border border-white/10 rounded-xl overflow-hidden">
      <button
        onClick={() => setExpanded(e => !e)}
        className="w-full flex items-center justify-between px-4 py-3 text-sm"
      >
        <span className="text-white/70">{icon} {category} <span className="text-white/30">({items.length} bills)</span></span>
        {expanded ? <ChevronUp size={13} className="text-white/30" /> : <ChevronDown size={13} className="text-white/30" />}
      </button>

      {expanded && (
        <div className="border-t border-white/10 divide-y divide-white/5">
          {items.map(t => (
            <TemplateEditRow key={t.id} template={t} onDelete={() => onDelete(t.id)} onSaved={onSaved} />
          ))}
        </div>
      )}
    </div>
  );
}

function TemplateEditRow({ template: t, onDelete, onSaved }: {
  template: FixedExpenseTemplate;
  onDelete: () => void;
  onSaved: () => void;
}) {
  const [name, setName]   = useState(t.name);
  const [amt, setAmt]     = useState(t.amount);
  const [dueDay, setDueDay] = useState(t.due_day ?? 0);
  const [saving, setSaving] = useState(false);

  const handleSave = async () => {
    setSaving(true);
    await api.put(`/fixed-templates/${t.id}`, {
      name, amount: amt, due_day: dueDay > 0 ? dueDay : null
    });
    onSaved(); setSaving(false);
  };

  return (
    <div className="flex items-center gap-2 px-3 py-2">
      <input
        value={name} onChange={e => setName(e.target.value)}
        className="flex-1 bg-dark-bg border border-white/10 rounded-lg px-2.5 py-1.5 text-white text-xs"
      />
      <input
        type="number" min="0" step="100"
        value={amt} onChange={e => setAmt(Number(e.target.value))}
        className="w-24 bg-dark-bg border border-white/10 rounded-lg px-2.5 py-1.5 text-white text-xs"
      />
      <select
        value={dueDay} onChange={e => setDueDay(Number(e.target.value))}
        className="w-28 bg-dark-bg border border-white/10 rounded-lg px-2 py-1.5 text-white text-xs"
      >
        <option value={0}>No reminder</option>
        {Array.from({ length: 28 }, (_, i) => i + 1).map(d => (
          <option key={d} value={d}>{d}th of month</option>
        ))}
      </select>
      <button onClick={handleSave} disabled={saving} className="text-indigo-400 hover:text-indigo-200 disabled:opacity-40">
        <Save size={13} />
      </button>
      <button onClick={onDelete} className="text-white/20 hover:text-red-400">
        <Trash2 size={13} />
      </button>
    </div>
  );
}
```

### `src/components/settings/CapsSection.tsx`

```typescript
import { useEffect, useState } from "react";
import { api } from "@/api/client";
import { useMonth } from "@/context/MonthContext";
import { CATEGORY_ICONS } from "@/utils/categories";
import { fmtInr } from "@/utils/formatInr";
import type { BudgetLimit, Summary } from "@/types";
import { Save } from "lucide-react";

export function CapsSection() {
  const { selMonth } = useMonth();
  const [budgets, setBudgets] = useState<BudgetLimit[]>([]);
  const [catSpent, setCatSpent] = useState<Record<string, number>>({});
  const [updates, setUpdates] = useState<Record<string, number>>({});
  const [saving, setSaving] = useState(false);
  const [saved, setSaved] = useState(false);

  useEffect(() => {
    api.get<BudgetLimit[]>("/budgets").then(r => {
      setBudgets(r.data);
      setUpdates(Object.fromEntries(r.data.map(b => [b.category, b.limit_amount])));
    });
    api.get<Summary>(`/summary/${selMonth}`).then(r => {
      setCatSpent(Object.fromEntries(r.data.categories.map(c => [c.category, c.spent])));
    }).catch(() => {});
  }, [selMonth]);

  const handleSave = async (e: React.FormEvent) => {
    e.preventDefault();
    setSaving(true);
    await Promise.all(
      Object.entries(updates).map(([category, limit_amount]) =>
        api.put("/budget", { category, limit_amount })
      )
    );
    setSaving(false); setSaved(true);
    setTimeout(() => setSaved(false), 2000);
  };

  return (
    <section>
      <div className="mb-4">
        <h2 className="font-syne font-bold text-white">🎯 Spending Caps</h2>
        <p className="text-white/40 text-sm mt-0.5">Monthly limit per category. You'll get a warning when you're close.</p>
        <div className="border-b border-white/10 mt-3" />
      </div>

      <form onSubmit={handleSave}>
        <div className="grid grid-cols-2 gap-4 mb-4">
          {budgets.map(b => {
            const spent = catSpent[b.category] ?? 0;
            const limit = updates[b.category] ?? b.limit_amount;
            const pct = limit > 0 ? (spent / limit) * 100 : 0;
            const colour = pct >= 100 ? "#ef4444" : pct >= 80 ? "#f59e0b" : "#34d399";

            return (
              <div key={b.category}>
                <div className="flex justify-between items-center mb-1">
                  <label className="text-white text-sm">{CATEGORY_ICONS[b.category] ?? "📦"} {b.category}</label>
                  <span className="text-xs" style={{ color: colour }}>{fmtInr(spent)} spent</span>
                </div>
                <input
                  type="number" min="0" step="500"
                  value={updates[b.category] ?? ""}
                  onChange={e => setUpdates(prev => ({ ...prev, [b.category]: Number(e.target.value) }))}
                  className="w-full bg-dark-card2 border border-white/10 rounded-xl px-3 py-2 text-white text-sm"
                />
              </div>
            );
          })}
        </div>
        <button
          type="submit" disabled={saving}
          className="flex items-center gap-2 bg-gradient-to-r from-accent to-accent2 text-white font-semibold px-6 py-2.5 rounded-xl text-sm disabled:opacity-50"
        >
          <Save size={14} /> {saving ? "Saving…" : saved ? "✅ Saved!" : "Save Spending Caps"}
        </button>
      </form>
    </section>
  );
}
```

### `src/components/settings/ShortcutsSection.tsx`

```typescript
import { useEffect, useState, useCallback } from "react";
import { api } from "@/api/client";
import { CATEGORY_ICONS, VAR_CATEGORIES } from "@/utils/categories";
import { fmtInr } from "@/utils/formatInr";
import type { ExpenseTemplate } from "@/types";
import { Trash2, Save, Plus } from "lucide-react";

export function ShortcutsSection() {
  const [shortcuts, setShortcuts] = useState<ExpenseTemplate[]>([]);
  const [newName, setNewName]     = useState("");
  const [newAmt, setNewAmt]       = useState(0);
  const [newCat, setNewCat]       = useState(VAR_CATEGORIES[0]);
  const [adding, setAdding]       = useState(false);

  const load = useCallback(() => {
    api.get<ExpenseTemplate[]>("/expense-templates").then(r => setShortcuts(r.data));
  }, []);
  useEffect(() => { load(); }, [load]);

  const handleAdd = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!newName.trim() || newAmt <= 0) return;
    setAdding(true);
    await api.post("/expense-templates", { name: newName, vendor: newName, category: newCat, amount: newAmt });
    setNewName(""); setNewAmt(0); setAdding(false);
    load();
  };

  const handleDelete = async (id: number) => {
    await api.delete(`/expense-templates/${id}`);
    setShortcuts(prev => prev.filter(s => s.id !== id));
  };

  const handleSave = async (t: ExpenseTemplate, updates: Partial<ExpenseTemplate>) => {
    await api.put(`/expense-templates/${t.id}`, updates);
    setShortcuts(prev => prev.map(s => s.id === t.id ? { ...s, ...updates } : s));
  };

  return (
    <section>
      <div className="mb-4">
        <h2 className="font-syne font-bold text-white">⚡ Saved Shortcuts</h2>
        <p className="text-white/40 text-sm mt-0.5">Expenses you log frequently. Appear as chips in Quick Add.</p>
        <div className="border-b border-white/10 mt-3" />
      </div>

      {shortcuts.length === 0 && (
        <p className="text-white/30 text-sm mb-4">No shortcuts yet. Add one below.</p>
      )}

      {/* Existing shortcuts */}
      {shortcuts.map(t => (
        <ShortcutEditRow key={t.id} template={t} onDelete={() => handleDelete(t.id)} onSave={(u) => handleSave(t, u)} />
      ))}
      {shortcuts.length > 0 && (
        <p className="text-white/30 text-xs mt-1 mb-4">{shortcuts.length} shortcut(s) · sorted by most used</p>
      )}

      {/* Add shortcut form */}
      <form onSubmit={handleAdd} className="grid grid-cols-3 gap-2">
        <input
          value={newName} onChange={e => setNewName(e.target.value)}
          placeholder="e.g. Petrol, Cook"
          className="bg-dark-card2 border border-white/10 rounded-xl px-3 py-2.5 text-white text-sm placeholder-white/30"
        />
        <input
          type="number" min="0" step="50"
          value={newAmt || ""}
          onChange={e => setNewAmt(Number(e.target.value))}
          placeholder="Amount (₹)"
          className="bg-dark-card2 border border-white/10 rounded-xl px-3 py-2.5 text-white text-sm"
        />
        <select
          value={newCat} onChange={e => setNewCat(e.target.value)}
          className="bg-dark-card2 border border-white/10 rounded-xl px-3 py-2.5 text-white text-sm"
        >
          {VAR_CATEGORIES.map(c => <option key={c} value={c}>{CATEGORY_ICONS[c]} {c}</option>)}
        </select>
        <button
          type="submit" disabled={adding || !newName.trim() || newAmt <= 0}
          className="col-span-3 flex items-center justify-center gap-2 bg-dark-card border border-white/10 text-white/70 hover:text-white py-2.5 rounded-xl text-sm disabled:opacity-40"
        >
          <Plus size={14} /> {adding ? "Adding…" : "Add Shortcut"}
        </button>
      </form>
    </section>
  );
}

function ShortcutEditRow({ template: t, onDelete, onSave }: {
  template: ExpenseTemplate;
  onDelete: () => void;
  onSave: (u: Partial<ExpenseTemplate>) => void;
}) {
  const [name, setName]   = useState(t.name);
  const [amt, setAmt]     = useState(t.amount);
  const [cat, setCat]     = useState(t.category);

  return (
    <div className="flex items-center gap-2 py-2 border-b border-white/5">
      <input value={name} onChange={e => setName(e.target.value)}
        className="flex-1 bg-dark-card2 border border-white/10 rounded-lg px-2.5 py-1.5 text-white text-xs" />
      <select value={cat} onChange={e => setCat(e.target.value)}
        className="w-28 bg-dark-card2 border border-white/10 rounded-lg px-2 py-1.5 text-white text-xs">
        {VAR_CATEGORIES.map(c => <option key={c} value={c}>{CATEGORY_ICONS[c]} {c}</option>)}
      </select>
      <input type="number" min="0" step="50"
        value={amt} onChange={e => setAmt(Number(e.target.value))}
        className="w-20 bg-dark-card2 border border-white/10 rounded-lg px-2.5 py-1.5 text-white text-xs" />
      <button onClick={() => onSave({ name, category: cat, amount: amt, vendor: name })}
        className="text-indigo-400 hover:text-indigo-200"><Save size={13} /></button>
      <button onClick={onDelete} className="text-white/20 hover:text-red-400"><Trash2 size={13} /></button>
    </div>
  );
}
```

### `src/components/settings/ExportSection.tsx`

```typescript
import { useState } from "react";
import { api } from "@/api/client";
import { useMonth } from "@/context/MonthContext";
import { fmtMonth } from "@/utils/formatDate";
import { Download, Loader2 } from "lucide-react";

export function ExportSection() {
  const { selMonth } = useMonth();
  const [loadingMonth, setLoadingMonth] = useState(false);
  const [loadingAll, setLoadingAll]     = useState(false);

  const download = async (url: string, filename: string, setLoading: (v: boolean) => void) => {
    setLoading(true);
    try {
      const { data } = await api.get(url, { responseType: "blob" });
      const href = URL.createObjectURL(data);
      const a = document.createElement("a");
      a.href = href; a.download = filename; a.click();
      URL.revokeObjectURL(href);
    } finally { setLoading(false); }
  };

  return (
    <section>
      <div className="mb-4">
        <h2 className="font-syne font-bold text-white">📥 My Data</h2>
        <p className="text-white/40 text-sm mt-0.5">Download your expense history as a CSV file.</p>
        <div className="border-b border-white/10 mt-3" />
      </div>

      <div className="flex gap-3">
        <button
          onClick={() => download(`/export/csv/${selMonth}`, `spendsense_${selMonth}.csv`, setLoadingMonth)}
          disabled={loadingMonth}
          className="flex-1 flex items-center justify-center gap-2 bg-dark-card2 border border-white/10 text-white/70 hover:text-white py-3 rounded-xl text-sm disabled:opacity-50"
        >
          {loadingMonth ? <Loader2 size={14} className="animate-spin" /> : <Download size={14} />}
          Download {fmtMonth(selMonth)}
        </button>
        <button
          onClick={() => download("/export/csv/all", `spendsense_all_${new Date().toISOString().slice(0,10)}.csv`, setLoadingAll)}
          disabled={loadingAll}
          className="flex-1 flex items-center justify-center gap-2 bg-dark-card2 border border-white/10 text-white/70 hover:text-white py-3 rounded-xl text-sm disabled:opacity-50"
        >
          {loadingAll ? <Loader2 size={14} className="animate-spin" /> : <Download size={14} />}
          Download Full History
        </button>
      </div>
    </section>
  );
}
```
