## Context

`frontend/app.py` (Streamlit) continues to run on `:8501` throughout.
This commit adds the Fixed tab to `frontend/react/` only.
Streamlit is the reference implementation — if behaviour is unclear, check
how `with tab2:` works in `frontend/app.py`.

The Fixed tab is a once-a-month workflow — users tick off bills as they pay them,
and add individual payments for variable recurring bills (pools).

> Theming: follow the CSS var pattern from T2.1. See T2.3 context for the full rule.

Key improvements over Streamlit:
- **Optimistic UI** — tick toggles instantly in the UI, syncs to backend in background.
  Streamlit does a full page rerun on every toggle, causing visible flicker.
- **Pool entries** — add-payment form is inline per pool, cleaner than Streamlit's
  `st.form()` which required a full rerun to show new entries.

### `src/components/tabs/FixedTab.tsx`

```typescript
import { useEffect, useState, useCallback } from "react";
import { api } from "@/api/client";
import { useMonth } from "@/context/MonthContext";
import { CATEGORY_ICONS } from "@/utils/categories";
import { fmtInr } from "@/utils/formatInr";
import type { Expense, Pool, PoolEntry } from "@/types";
import { Bell, Plus, Trash2, ChevronDown, ChevronUp } from "lucide-react";

interface DueReminder { vendor: string; amount: number; due_day: number; days_overdue: number; }

export function FixedTab() {
  const { selMonth, isCurrent } = useMonth();

  const [reminders, setReminders]   = useState<DueReminder[]>([]);
  const [fixedExps, setFixedExps]   = useState<Expense[]>([]);
  const [pools, setPools]           = useState<Pool[]>([]);
  const [loading, setLoading]       = useState(true);

  const load = useCallback(async () => {
    setLoading(true);
    const [rem, fixed, poolData] = await Promise.all([
      isCurrent ? api.get<DueReminder[]>(`/fixed/due-reminders/${selMonth}`).then(r => r.data) : Promise.resolve([]),
      api.get<Expense[]>(`/fixed/${selMonth}`).then(r => r.data),
      api.get<Pool[]>(`/pools/${selMonth}`).then(r => r.data),
    ]);
    setReminders(rem);
    setFixedExps(fixed);
    setPools(poolData);
    setLoading(false);
  }, [selMonth, isCurrent]);

  useEffect(() => { load(); }, [load]);

  // Optimistic toggle — update UI immediately, sync backend in background
  const togglePaid = async (id: number) => {
    setFixedExps(prev =>
      prev.map(e => e.id === id ? { ...e, paid: !e.paid } : e)
    );
    await api.patch(`/fixed/${id}/toggle`);
  };

  if (loading) return <FixedTabSkeleton />;

  const paidTotal   = fixedExps.reduce((s, e) => s + (e.paid ? e.amount : 0), 0);
  const unpaidTotal = fixedExps.reduce((s, e) => s + (!e.paid ? e.amount : 0), 0);
  const totalFixed  = paidTotal + unpaidTotal;
  const paidCount   = fixedExps.filter(e => e.paid).length;
  const pct         = totalFixed > 0 ? Math.round(paidTotal / totalFixed * 100) : 0;

  // Group by category
  const byCategory = fixedExps.reduce<Record<string, Expense[]>>((acc, e) => {
    (acc[e.category] ??= []).push(e);
    return acc;
  }, {});

  const isEmpty = fixedExps.length === 0 && pools.length === 0;

  return (
    <div className="space-y-6">

      {/* ── Due Reminders ───────────────────────────────── */}
      {reminders.map((r, i) => (
        <div key={i} className="flex items-center gap-3 p-3 rounded-xl bg-red-500/10 border-l-4 border-red-500 text-red-300 text-sm">
          <Bell size={14} className="flex-shrink-0" />
          <span>
            <b>{r.vendor}</b> {fmtInr(r.amount)} — was due on the {r.due_day}th
            {r.days_overdue === 0 ? " (due today)" : ` (${r.days_overdue}d overdue)`}
          </span>
        </div>
      ))}

      {/* ── Fixed Expenses ──────────────────────────────── */}
      {fixedExps.length > 0 && (
        <section>
          <div className="flex justify-between text-sm mb-2">
            <span className="text-white/40">{paidCount} of {fixedExps.length} paid · {fmtInr(paidTotal)} done</span>
            <span className="text-red-400">{fmtInr(unpaidTotal)} pending</span>
          </div>

          {/* Progress bar */}
          <div className="h-1.5 bg-white/5 rounded-full mb-5 overflow-hidden">
            <div
              className="h-full rounded-full transition-all duration-500"
              style={{ width: `${pct}%`, background: "linear-gradient(90deg, #34d399, #6366f1)" }}
            />
          </div>

          {Object.entries(byCategory).sort().map(([cat, items]) => (
            <div key={cat} className="mb-5">
              <p className="text-xs font-syne font-bold uppercase tracking-widest text-white/30 mb-2">
                {CATEGORY_ICONS[cat] ?? "📦"} {cat} · {fmtInr(items.reduce((s, e) => s + e.amount, 0))}
              </p>
              <div className="space-y-1">
                {items.map(item => (
                  <FixedExpenseRow
                    key={item.id}
                    item={item}
                    onToggle={() => togglePaid(item.id)}
                  />
                ))}
              </div>
            </div>
          ))}
        </section>
      )}

      {/* ── Essential Pools ─────────────────────────────── */}
      {pools.length > 0 && (
        <section>
          <h2 className="text-xs font-syne font-bold uppercase tracking-widest text-white/40 mb-1">
            Essential Pools
          </h2>
          <p className="text-white/30 text-xs mb-4">
            Bills with variable amount — add each payment as it happens.
          </p>
          <div className="space-y-3">
            {pools.map(pool => (
              <PoolCard
                key={pool.id}
                pool={pool}
                selMonth={selMonth}
                onChanged={load}
              />
            ))}
          </div>
        </section>
      )}

      {/* ── Empty state ─────────────────────────────────── */}
      {isEmpty && (
        <div className="text-center py-12">
          <p className="text-white/30 text-sm">
            No bills set up yet.<br />
            <span className="text-indigo-400">Go to Settings → Monthly Bills</span>{" "}
            to add your rent, EMI, and subscriptions.
          </p>
        </div>
      )}
    </div>
  );
}
```

### `src/components/tabs/PoolCard.tsx`

```typescript
import { useState } from "react";
import { api } from "@/api/client";
import { CATEGORY_ICONS } from "@/utils/categories";
import { fmtInr } from "@/utils/formatInr";
import type { Pool, PoolEntry } from "@/types";
import { Plus, Trash2, ChevronDown, ChevronUp } from "lucide-react";

interface Props { pool: Pool; selMonth: string; onChanged: () => void; }

export function PoolCard({ pool, selMonth, onChanged }: Props) {
  const [expanded, setExpanded] = useState(true);
  const [label, setLabel] = useState("");
  const [amount, setAmount] = useState(0);
  const [adding, setAdding] = useState(false);
  const icon = CATEGORY_ICONS[pool.category] ?? "📦";

  const addEntry = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!label.trim() || amount <= 0) return;
    setAdding(true);
    try {
      await api.post(`/pools/${pool.id}/entries/${selMonth}`, { label, amount });
      setLabel(""); setAmount(0);
      onChanged();
    } finally { setAdding(false); }
  };

  const deleteEntry = async (entryId: number) => {
    await api.delete(`/pools/entries/${entryId}`);
    onChanged();
  };

  const toggleEntry = async (entryId: number) => {
    await api.patch(`/pools/entries/${entryId}/toggle`);
    onChanged();
  };

  const poolStatus =
    pool.entry_count === 0 ? "⚠️ No entries yet"
    : pool.unpaid_total === 0 ? `✅ ${fmtInr(pool.paid_total)} paid`
    : `${fmtInr(pool.paid_total)} paid · ${fmtInr(pool.unpaid_total)} unpaid`;

  return (
    <div className="bg-dark-card border border-white/10 rounded-2xl overflow-hidden">
      {/* Pool header */}
      <button
        onClick={() => setExpanded(e => !e)}
        className="w-full flex items-center justify-between px-4 py-3"
      >
        <span className="font-semibold text-white text-sm">{icon} {pool.name}</span>
        <div className="flex items-center gap-2">
          <span className="text-white/40 text-xs">{poolStatus}</span>
          {expanded ? <ChevronUp size={14} className="text-white/30" /> : <ChevronDown size={14} className="text-white/30" />}
        </div>
      </button>

      {expanded && (
        <div className="px-4 pb-4 space-y-2">
          {/* Existing entries */}
          {pool.entries.map(entry => (
            <div key={entry.id} className="flex items-center gap-3 py-1.5 border-b border-white/5">
              <button
                onClick={() => toggleEntry(entry.id)}
                className={`w-5 h-5 rounded-full border-2 flex items-center justify-center flex-shrink-0 ${
                  entry.paid ? "bg-emerald-500 border-emerald-500 text-white" : "border-white/20"
                }`}
              >
                {entry.paid && <span className="text-[10px]">✓</span>}
              </button>
              <span className={`flex-1 text-sm ${entry.paid ? "line-through text-white/30" : "text-white"}`}>
                {entry.label}
                {entry.note && <span className="text-white/30 ml-2 text-xs">· {entry.note}</span>}
              </span>
              <span className={`font-syne font-semibold text-sm ${entry.paid ? "text-emerald-400" : "text-white/50"}`}>
                {fmtInr(entry.amount)}
              </span>
              <button
                onClick={() => deleteEntry(entry.id)}
                className="text-white/20 hover:text-red-400 ml-1"
              >
                <Trash2 size={13} />
              </button>
            </div>
          ))}

          {/* Add payment form */}
          <form onSubmit={addEntry} className="flex gap-2 pt-2">
            <input
              value={label} onChange={e => setLabel(e.target.value)}
              placeholder="Label (e.g. Home, Self)"
              className="flex-1 bg-dark-card2 border border-white/10 rounded-lg px-3 py-2 text-white text-sm placeholder-white/30"
            />
            <input
              type="number" min="0" step="10"
              value={amount || ""}
              onChange={e => setAmount(Number(e.target.value))}
              placeholder="₹"
              className="w-24 bg-dark-card2 border border-white/10 rounded-lg px-3 py-2 text-white text-sm"
            />
            <button
              type="submit" disabled={adding || !label.trim() || amount <= 0}
              className="flex items-center gap-1 bg-accent/20 hover:bg-accent/40 text-indigo-300 px-3 py-2 rounded-lg text-sm disabled:opacity-40"
            >
              <Plus size={14} /> Add
            </button>
          </form>

          {/* Pool total */}
          {pool.paid_total + pool.unpaid_total > 0 && (
            <div className="flex justify-between pt-2 border-t border-white/5 text-xs">
              <span className="text-white/30">{pool.entry_count} payment(s)</span>
              <span className="text-white font-syne font-bold">
                {fmtInr(pool.paid_total + pool.unpaid_total)} total ·{" "}
                <span className="text-emerald-400">{fmtInr(pool.paid_total)} paid</span>
              </span>
            </div>
          )}
        </div>
      )}
    </div>
  );
}
```
